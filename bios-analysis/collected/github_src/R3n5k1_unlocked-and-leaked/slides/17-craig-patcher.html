STEP 9 :: AUTOMATE THE PATCH

With this information, we can vibe-code a patcher to find the pattern and disable pre-boot DMA in the flash dump, ready to reflash.

import os, re # The Setup NVRAM variable: GUID + UTF-16 name "Setup" SETUP_GUID = b'\x43\xd6\x87\xec\xa4\xeb\xb5\x4b\xa1\xe5\x3f\x3e\x36\xb2\x0d\xa9' SETUP_NAME = b'S\x00e\x00t\x00u\x00p\x00\x00\x00' # UTF-16 "Setup\0" VAR_OFFSET = 0x975 # Control Iommu Pre-boot Behavior HEADER_SIZE = 16 + 12 # GUID (16) + Name (12) with open(filename, 'rb') as f: data = bytearray(f.read()) # Scan for all occurrences of the Setup GUID matches = [m.start() for m in re.finditer(re.escape(SETUP_GUID), data)] for start in matches: if data[start+16 : start+28] == SETUP_NAME: patch_addr = start + HEADER_SIZE + VAR_OFFSET print(f" [+] Patching offset 0x{patch_addr:x}: {data[patch_addr]:#04x} -> 0x00") data[patch_addr] = 0x00 # disable pre-boot DMA protection with open(out_file, 'wb') as f: f.write(data) print(f"Saved patched dump: {out_file}")
